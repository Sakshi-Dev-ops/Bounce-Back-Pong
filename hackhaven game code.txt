<canvas id="gameCanvas" width="900" height="600"></canvas>

<script>
const canvas = document.getElementById("gameCanvas");
const ctx = canvas.getContext("2d");

const W = 900;
const H = 600;

// Game States
const SCREENS = {
    WELCOME: 0,
    RULES: 1,
    MODE_SELECT: 2,
    CONTROL_SELECT: 3,
    PRE_START: 4,
    TUTORIAL: 5,
    COUNTDOWN: 6,
    GAMEPLAY: 7,
    GAMEOVER: 8
};

let currentScreen = SCREENS.WELCOME;

// Settings & State
let isSolo = true;
let useMouse = false;
let countdownTimer = 3.0;

let score1 = 0;
let score2 = 0;
let lastSpeedMilestone = 0;
let winnerText = "";

// Paddles
const paddleW = 15;
const paddleH = 100;
const paddleSpeed = 420;
let leftPaddle = { x: 30, y: H / 2 - paddleH / 2, w: paddleW, h: paddleH };
let rightPaddle = { x: W - 30 - paddleW, y: H / 2 - paddleH / 2, w: paddleW, h: paddleH };

// Ball
let ball = { x: W / 2, y: H / 2, vx: 350, vy: 220, r: 10 };
let speedMultiplier = 1.0;

// Interactive Tutorial Mini-Arena
const demoBox = { x: 150, y: 145, w: 600, h: 240 };
let demoLeft = { x: demoBox.x + 20, y: demoBox.y + 85, w: 12, h: 70 };
let demoRight = { x: demoBox.x + demoBox.w - 32, y: demoBox.y + 85, w: 12, h: 70 };
let demoBall = { x: demoBox.x + demoBox.w / 2, y: demoBox.y + demoBox.h / 2, vx: 240, vy: 160, r: 8 };

// Input Tracking
const keys = {};
let mousePos = { x: 0, y: H / 2 };
let mouseClicked = false;

window.addEventListener("keydown", (e) => {
    keys[e.code] = true;
    if (["ArrowUp", "ArrowDown", "Space"].includes(e.code)) e.preventDefault();
});
window.addEventListener("keyup", (e) => { keys[e.code] = false; });

canvas.addEventListener("mousemove", (e) => {
    const rect = canvas.getBoundingClientRect();
    const scaleX = W / rect.width;
    const scaleY = H / rect.height;
    mousePos.x = (e.clientX - rect.left) * scaleX;
    mousePos.y = (e.clientY - rect.top) * scaleY;
});

canvas.addEventListener("mousedown", (e) => {
    if (e.button === 0) mouseClicked = true;
});

function resetDemo() {
    demoLeft.y = demoBox.y + (demoBox.h - demoLeft.h) / 2;
    demoRight.y = demoBox.y + (demoBox.h - demoRight.h) / 2;
    demoBall.x = demoBox.x + demoBox.w / 2;
    demoBall.y = demoBox.y + demoBox.h / 2;
    demoBall.vx = 240;
    demoBall.vy = 160;
}

function resetMatch() {
    score1 = 0;
    score2 = 0;
    lastSpeedMilestone = 0;
    speedMultiplier = 1.0;
    leftPaddle.y = H / 2 - paddleH / 2;
    rightPaddle.y = H / 2 - paddleH / 2;
    ball.x = W / 2;
    ball.y = H / 2;
    ball.vx = 350;
    ball.vy = 220;
    countdownTimer = 3.0;
}

function circleRectCollide(cx, cy, cr, rect) {
    let closestX = Math.max(rect.x, Math.min(cx, rect.x + rect.w));
    let closestY = Math.max(rect.y, Math.min(cy, rect.y + rect.h));
    let dx = cx - closestX;
    let dy = cy - closestY;
    return (dx * dx + dy * dy) <= (cr * cr);
}

function drawButton(x, y, w, h, text, baseColor, hoverColor) {
    const isHover = mousePos.x >= x && mousePos.x <= x + w && mousePos.y >= y && mousePos.y <= y + h;
    ctx.fillStyle = isHover ? hoverColor : baseColor;
    ctx.fillRect(x, y, w, h);
    ctx.strokeStyle = "#ffffff";
    ctx.lineWidth = 2;
    ctx.strokeRect(x, y, w, h);

    ctx.fillStyle = "#ffffff";
    ctx.font = "bold 22px Arial";
    ctx.textAlign = "center";
    ctx.textBaseline = "middle";
    ctx.fillText(text, x + w / 2, y + h / 2);

    return isHover && mouseClicked;
}

function drawCenteredText(text, y, size, color, bold = true) {
    ctx.fillStyle = color;
    ctx.font = `${bold ? "bold " : ""}${size}px Arial`;
    ctx.textAlign = "center";
    ctx.textBaseline = "alphabetic";
    ctx.fillText(text, W / 2, y);
}

// Main Game Loop
let lastTime = performance.now();

function gameLoop(now) {
    const dt = Math.min((now - lastTime) / 1000, 0.05);
    lastTime = now;

    // UPDATE
    if (currentScreen === SCREENS.TUTORIAL) {
        if (isSolo && useMouse) {
            demoLeft.y = mousePos.y - demoLeft.h / 2;
        } else {
            if (keys["KeyW"]) demoLeft.y -= 320 * dt;
            if (keys["KeyS"]) demoLeft.y += 320 * dt;
        }
        demoLeft.y = Math.max(demoBox.y + 5, Math.min(demoBox.y + demoBox.h - 5 - demoLeft.h, demoLeft.y));

        if (!isSolo) {
            if (keys["ArrowUp"]) demoRight.y -= 320 * dt;
            if (keys["ArrowDown"]) demoRight.y += 320 * dt;
            demoRight.y = Math.max(demoBox.y + 5, Math.min(demoBox.y + demoBox.h - 5 - demoRight.h, demoRight.y));
        }

        demoBall.x += demoBall.vx * dt;
        demoBall.y += demoBall.vy * dt;

        if (demoBall.y - demoBall.r <= demoBox.y) {
            demoBall.y = demoBox.y + demoBall.r;
            demoBall.vy *= -1;
        }
        if (demoBall.y + demoBall.r >= demoBox.y + demoBox.h) {
            demoBall.y = demoBox.y + demoBox.h - demoBall.r;
            demoBall.vy *= -1;
        }

        if (circleRectCollide(demoBall.x, demoBall.y, demoBall.r, demoLeft)) {
            demoBall.x = demoLeft.x + demoLeft.w + demoBall.r;
            demoBall.vx = Math.abs(demoBall.vx);
        }

        if (isSolo) {
            if (demoBall.x + demoBall.r >= demoBox.x + demoBox.w - 10) {
                demoBall.x = demoBox.x + demoBox.w - 10 - demoBall.r;
                demoBall.vx = -Math.abs(demoBall.vx);
            }
        } else {
            if (circleRectCollide(demoBall.x, demoBall.y, demoBall.r, demoRight)) {
                demoBall.x = demoRight.x - demoBall.r;
                demoBall.vx = -Math.abs(demoBall.vx);
            }
        }

        if (demoBall.x < demoBox.x || demoBall.x > demoBox.x + demoBox.w) {
            demoBall.x = demoBox.x + demoBox.w / 2;
            demoBall.y = demoBox.y + demoBox.h / 2;
            demoBall.vx *= -1;
        }
    }
    else if (currentScreen === SCREENS.COUNTDOWN) {
        countdownTimer -= dt;
        if (countdownTimer <= 0) currentScreen = SCREENS.GAMEPLAY;
    }
    else if (currentScreen === SCREENS.GAMEPLAY) {
        if (isSolo && useMouse) {
            leftPaddle.y = mousePos.y - leftPaddle.h / 2;
        } else {
            if (keys["KeyW"]) leftPaddle.y -= paddleSpeed * dt;
            if (keys["KeyS"]) leftPaddle.y += paddleSpeed * dt;
        }
        leftPaddle.y = Math.max(50, Math.min(H - leftPaddle.h, leftPaddle.y));

        if (!isSolo) {
            if (keys["ArrowUp"]) rightPaddle.y -= paddleSpeed * dt;
            if (keys["ArrowDown"]) rightPaddle.y += paddleSpeed * dt;
            rightPaddle.y = Math.max(50, Math.min(H - rightPaddle.h, rightPaddle.y));
        }

        ball.x += ball.vx * speedMultiplier * dt;
        ball.y += ball.vy * speedMultiplier * dt;

        if (ball.y - ball.r <= 50) {
            ball.y = 50 + ball.r;
            ball.vy *= -1;
        }
        if (ball.y + ball.r >= H) {
            ball.y = H - ball.r;
            ball.vy *= -1;
        }

        if (circleRectCollide(ball.x, ball.y, ball.r, leftPaddle)) {
            ball.x = leftPaddle.x + leftPaddle.w + ball.r;
            ball.vx = Math.abs(ball.vx);

            if (isSolo) {
                score1 += 10;
                if (Math.floor(score1 / 30) > lastSpeedMilestone) {
                    lastSpeedMilestone = Math.floor(score1 / 30);
                    speedMultiplier += 0.15;
                }
                if (score1 >= 100) {
                    winnerText = "YOU REACHED 100 POINTS! YOU WIN!";
                    currentScreen = SCREENS.GAMEOVER;
                }
            }
        }

        if (isSolo) {
            if (ball.x + ball.r >= W - 15) {
                ball.x = W - 15 - ball.r;
                ball.vx = -Math.abs(ball.vx);
            }
            if (ball.x < 0) {
                winnerText = "GAME OVER! Final Score: " + score1;
                currentScreen = SCREENS.GAMEOVER;
            }
        } else {
            if (circleRectCollide(ball.x, ball.y, ball.r, rightPaddle)) {
                ball.x = rightPaddle.x - ball.r;
                ball.vx = -Math.abs(ball.vx);
            }

            if (ball.x < 0) {
                score2++;
                if (score2 % 2 === 0) speedMultiplier += 0.15;
                if (score2 >= 10) {
                    winnerText = "PLAYER 2 WINS!";
                    currentScreen = SCREENS.GAMEOVER;
                } else {
                    ball.x = W / 2;
                    ball.y = H / 2;
                    ball.vx = 350;
                }
            }

            if (ball.x > W) {
                score1++;
                if (score1 % 2 === 0) speedMultiplier += 0.15;
                if (score1 >= 10) {
                    winnerText = "PLAYER 1 WINS!";
                    currentScreen = SCREENS.GAMEOVER;
                } else {
                    ball.x = W / 2;
                    ball.y = H / 2;
                    ball.vx = -350;
                }
            }
        }
    }

    // DRAW
    ctx.fillStyle = "#121621";
    ctx.fillRect(0, 0, W, H);

    if (currentScreen === SCREENS.WELCOME) {
        drawCenteredText("BOUNCE BACK PONG", 95, 46, "#38bdf8");
        drawCenteredText("Created by Sakshi.M and Pari.P", 145, 24, "#facc15");

        ctx.strokeStyle = "#475569";
        ctx.lineWidth = 2;
        ctx.strokeRect(80, 185, W - 160, 220);

        ctx.textAlign = "left";
        ctx.fillStyle = "#ffffff";
        ctx.font = "bold 20px Arial";
        ctx.fillText("HOW IT RELATES TO THE THEME 'BOUNCE BACK':", 110, 225);

        ctx.fillStyle = "#cbd5e1";
        ctx.font = "18px Arial";
        ctx.fillText("• Every shot coming at you is a chance to bounce back stronger!", 110, 265);
        ctx.fillText("• As your rally continues, the ball speeds up, testing your reflexes", 110, 300);
        ctx.fillText("  and forcing you to adapt and bounce back under pressure.", 110, 328);
        ctx.fillText("• Never let the ball slip past—keep the momentum alive!", 110, 368);

        if (drawButton(W / 2 - 120, 450, 240, 55, "NEXT: RULES", "#1d4ed8", "#2563eb")) {
            currentScreen = SCREENS.RULES;
        }
    }
    else if (currentScreen === SCREENS.RULES) {
        drawCenteredText("GAME RULES", 85, 42, "#38bdf8");

        ctx.textAlign = "left";
        ctx.fillStyle = "#facc15";
        ctx.font = "bold 24px Arial";
        ctx.fillText("SOLO MODE:", 100, 155);

        ctx.fillStyle = "#ffffff";
        ctx.font = "19px Arial";
        ctx.fillText("• Bounce the ball off the back wall to earn +10 points per hit.", 120, 190);
        ctx.fillText("• Every 30 points, the game gets faster!", 120, 222);
        ctx.fillStyle = "#4ade80";
        ctx.fillText("• Reach 100 points to WIN the game!", 120, 254);

        ctx.fillStyle = "#facc15";
        ctx.font = "bold 24px Arial";
        ctx.fillText("2-PLAYER MODE:", 100, 315);

        ctx.fillStyle = "#ffffff";
        ctx.font = "19px Arial";
        ctx.fillText("• Player 1 (Left) uses W / S. Player 2 (Right) uses UP / DOWN Arrows.", 120, 350);
        ctx.fillText("• Every 2 points a player earns, the ball speeds up!", 120, 382);
        ctx.fillStyle = "#4ade80";
        ctx.fillText("• First player to 10 points wins the match!", 120, 414);

        if (drawButton(W / 2 - 140, 470, 280, 55, "CHOOSE GAME MODE", "#1d4ed8", "#2563eb")) {
            currentScreen = SCREENS.MODE_SELECT;
        }
    }
    else if (currentScreen === SCREENS.MODE_SELECT) {
        drawCenteredText("SELECT GAME MODE", 125, 40, "#38bdf8");

        if (drawButton(W / 2 - 160, 220, 320, 65, "1 PLAYER (SOLO)", "#15803d", "#16a34a")) {
            isSolo = true;
            currentScreen = SCREENS.CONTROL_SELECT;
        }

        if (drawButton(W / 2 - 160, 320, 320, 65, "2 PLAYERS (VERSUS)", "#991b1b", "#dc2626")) {
            isSolo = false;
            useMouse = false;
            currentScreen = SCREENS.PRE_START;
        }
    }
    else if (currentScreen === SCREENS.CONTROL_SELECT) {
        drawCenteredText("CHOOSE YOUR CONTROLS", 120, 38, "#38bdf8");
        drawCenteredText("(Mouse control is exclusively available for Solo Mode)", 165, 19, "#94a3b8", false);

        if (drawButton(W / 2 - 160, 230, 320, 65, "KEYBOARD (W / S)", "#1d4ed8", "#2563eb")) {
            useMouse = false;
            currentScreen = SCREENS.PRE_START;
        }

        if (drawButton(W / 2 - 160, 330, 320, 65, "MOUSE CONTROL", "#6b21a8", "#7e22ce")) {
            useMouse = true;
            currentScreen = SCREENS.PRE_START;
        }
    }
    else if (currentScreen === SCREENS.PRE_START) {
        drawCenteredText("READY TO PLAY?", 120, 42, "#38bdf8");

        const summary = isSolo
            ? (useMouse ? "Selected: 1 Player (Mouse Control)" : "Selected: 1 Player (Keyboard: W / S)")
            : "Selected: 2 Players (P1: W/S | P2: Up/Down Arrows)";
        drawCenteredText(summary, 175, 22, "#facc15");

        if (drawButton(W / 2 - 170, 235, 340, 65, "VIEW TUTORIAL (DEMO)", "#1d4ed8", "#2563eb")) {
            resetDemo();
            currentScreen = SCREENS.TUTORIAL;
        }

        if (drawButton(W / 2 - 170, 330, 340, 65, "START GAME", "#15803d", "#16a34a")) {
            resetMatch();
            currentScreen = SCREENS.COUNTDOWN;
        }
    }
    else if (currentScreen === SCREENS.TUTORIAL) {
        drawCenteredText("INTERACTIVE TUTORIAL - TRY IT OUT!", 60, 34, "#facc15");

        let hint = "";
        if (isSolo && useMouse) hint = "Move your MOUSE up & down to bounce the ball off the right wall!";
        else if (isSolo && !useMouse) hint = "Press [W] (Up) and [S] (Down) to bounce the ball off the right wall!";
        else hint = "Player 1: Press [W]/[S]  |  Player 2: Press [UP]/[DOWN] Arrows!";
        drawCenteredText(hint, 105, 19, "#ffffff", false);

        ctx.fillStyle = "#0a0d16";
        ctx.fillRect(demoBox.x, demoBox.y, demoBox.w, demoBox.h);
        ctx.strokeStyle = "#38bdf8";
        ctx.lineWidth = 3;
        ctx.strokeRect(demoBox.x, demoBox.y, demoBox.w, demoBox.h);

        ctx.fillStyle = "#38bdf8";
        ctx.fillRect(demoLeft.x, demoLeft.y, demoLeft.w, demoLeft.h);

        if (isSolo) {
            ctx.fillStyle = "#475569";
            ctx.fillRect(demoBox.x + demoBox.w - 10, demoBox.y, 10, demoBox.h);
        } else {
            ctx.fillStyle = "#fb923c";
            ctx.fillRect(demoRight.x, demoRight.y, demoRight.w, demoRight.h);
        }

        ctx.fillStyle = "#ffffff";
        ctx.beginPath();
        ctx.arc(demoBall.x, demoBall.y, demoBall.r, 0, Math.PI * 2);
        ctx.fill();

        if (drawButton(W / 2 - 110, 405, 220, 40, "RESET DEMO BALL", "#334155", "#475569")) {
            resetDemo();
        }

        if (drawButton(W / 2 - 150, 470, 300, 65, "START GAME", "#15803d", "#16a34a")) {
            resetMatch();
            currentScreen = SCREENS.COUNTDOWN;
        }
    }
    else if (currentScreen === SCREENS.COUNTDOWN) {
        const countNum = Math.ceil(countdownTimer);
        drawCenteredText(`Game starts in ${countNum}!`, H / 2 + 12, 46, "#facc15");
    }
    else if (currentScreen === SCREENS.GAMEPLAY) {
        // Top HUD
        ctx.fillStyle = "#0a0c14";
        ctx.fillRect(0, 0, W, 50);
        ctx.strokeStyle = "#38bdf8";
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(0, 50);
        ctx.lineTo(W, 50);
        ctx.stroke();

        // Score (Top-Left)
        ctx.textAlign = "left";
        ctx.textBaseline = "middle";
        ctx.fillStyle = "#ffffff";
        ctx.font = "bold 22px Arial";
        if (isSolo) {
            ctx.fillText(`SCORE: ${score1} / 100`, 20, 26);
        } else {
            ctx.fillText(`P1: ${score1}   |   P2: ${score2}`, 20, 26);
        }

        // Creator Credits (Top-Right)
        ctx.textAlign = "right";
        ctx.fillStyle = "#facc15";
        ctx.font = "bold 19px Arial";
        ctx.fillText("Made by Sakshi.M & Pari.P", W - 20, 26);

        // Left Paddle
        ctx.fillStyle = "#38bdf8";
        ctx.fillRect(leftPaddle.x, leftPaddle.y, leftPaddle.w, leftPaddle.h);

        // Right Wall or Paddle
        if (isSolo) {
            ctx.fillStyle = "#475569";
            ctx.fillRect(W - 15, 50, 15, H - 50);
        } else {
            ctx.fillStyle = "#fb923c";
            ctx.fillRect(rightPaddle.x, rightPaddle.y, rightPaddle.w, rightPaddle.h);
            ctx.fillStyle = "#334155";
            for (let y = 60; y < H; y += 30) {
                ctx.fillRect(W / 2 - 2, y, 4, 15);
            }
        }

        // Ball
        ctx.fillStyle = "#ffffff";
        ctx.beginPath();
        ctx.arc(ball.x, ball.y, ball.r, 0, Math.PI * 2);
        ctx.fill();
    }
    else if (currentScreen === SCREENS.GAMEOVER) {
        drawCenteredText(winnerText, 210, 36, "#facc15");
        drawCenteredText("Thanks for playing Bounce Back Pong!", 265, 22, "#cbd5e1", false);

        if (drawButton(W / 2 - 150, 330, 300, 65, "PLAY AGAIN (RESTART)", "#15803d", "#16a34a")) {
            resetMatch();
            currentScreen = SCREENS.MODE_SELECT;
        }
    }

    mouseClicked = false;
    requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
</script>
</body>
